78: def verify_cash_structure(s, cash):
79:     """Prove lot classification, original short proceeds and settlement restrictions.
80: 
81:     Settlement uses the durable Journal's completed session, not a mutable lot flag
82:     or the scheduler's current session. Negative available cash alone is not an
83:     inconsistency: a correctly recorded margin breach must keep its actual loss.
84:     """
85:     sessions = []
86:     for event in s["journal"]:
87:         if event["kind"] == "SETTLEMENT":
88:             value = event.get("settlement_session")
89:             if type(value) is not int or value < 0 or (sessions and value < sessions[-1]):
90:                 raise ContractError("UNVERIFIED_SETTLEMENT_SESSION")
91:             sessions.append(value)
92:     settled_through = sessions[-1] if sessions else -1
93:     backing = {}
94:     for asset in s["assets"].values():
95:         for lot in asset["lots"]:
96:             if lot["direction"] == -1:
97:                 if lot["id"] in backing or lot["remaining"] < 0 or lot["price"] < 0:
98:                     raise ContractError("INVALID_SHORT_PROCEEDS_BACKING")
99:                 backing[lot["id"]] = lot
100:     ids, bound = set(), set()
101:     for lot in s["cashlots"]:
102:         if (
103:             set(lot) != {"id", "entry_id", "kind", "amount", "settled", "due"}
104:             or not isinstance(lot["id"], str)
105:             or not lot["id"]
106:             or lot["id"] in ids
107:             or lot["kind"] not in {"FREE", "SHORT"}
108:             or lot["amount"] < 0
109:             or type(lot["settled"]) is not bool
110:             or type(lot["due"]) is not int
111:             or lot["due"] < 0
112:             or lot["settled"] != (lot["due"] <= settled_through)
113:         ):
114:             raise ContractError("INVALID_CASH_LOT_CLASSIFICATION")
115:         ids.add(lot["id"])
116:         if lot["kind"] == "FREE":
117:             if lot["entry_id"] is not None:
118:                 raise ContractError("INVALID_FREE_CASH_BACKING")
119:         else:
120:             entry = lot["entry_id"]
121:             if entry not in backing or entry in bound:
122:                 raise ContractError("INVALID_SHORT_PROCEEDS_BACKING")
123:             original = backing[entry]
124:             if (
125:                 lot["amount"] != original["remaining"] * original["price"]
126:                 or lot["due"] != original["settles"]
127:             ):
128:                 raise ContractError("SHORT_PROCEEDS_OR_SETTLEMENT_MISMATCH")
129:             bound.add(entry)
130:     if bound != set(backing):
131:         raise ContractError("MISSING_SHORT_PROCEEDS_BACKING")
132:     if any(cash[k] < 0 for k in ("U_free", "U_short", "S_set")) or cash["S_set"] > cash["cash_set"]:
133:         raise ContractError("RESTRICTED_CASH_SUBSET_FAILED")
134:     return {
135:         "status": "EXACT_CASH_LOTS_AND_SHORT_BACKING_VERIFIED",
136:         "cash_lots": len(ids),
137:         "short_entries": len(bound),
138:         "settled_through_session": settled_through,
139:     }
