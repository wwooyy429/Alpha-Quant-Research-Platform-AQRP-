40: def read_core(binding):
41:     """Read-only terminal proof including original bytes and pre-call initial binding."""
42:     root = guard_path(binding["physical_ledger_root"])

209:     def status(self, task):
210:         row = self._snapshot(task)
211:         public = {
212:             "task_id": task,
213:             "config_hash": row["hash"],
214:             "status": row["status"],
215:             "cancel_requested": bool(row["cancel"]),
216:             "error": row["error"],
217:             "slots": self.state.slots(task),
218:             "scope": "ENGINEERING_TEST_ONLY",
219:             "financial_approval": "NONE",
220:         }
221:         if "campaign" in load(row["raw"].encode()).get("payload", {}):
222:             public["research_control"] = self.state.research_attempt(task)
223:         if row["status"] == "CANCELLED":
224:             try:
225:                 self._verify_cancelled_finances(task)
226:             except (Exception, KeyboardInterrupt) as exc:
227:                 public.update(
228:                     status="INTEGRITY_ERROR",
229:                     usable=False,
230:                     error="COMMITTED_STATE_UNVERIFIED",
231:                     financial_verification_error=getattr(exc, "code", type(exc).__name__),
232:                     original_terminal_status="CANCELLED",
233:                 )
234:         if row["status"] == "SUCCEEDED":
235:             try:
236:                 self.validate_effective(row["raw"].encode())
237:                 pointer = load(row["completion"])
238:                 if set(pointer) != {"file", "sha256"} or not re.fullmatch(
239:                     r"completion(?:-[0-9]+)?\.json", pointer["file"]
240:                 ):
241:                     raise ContractError("TASK_COMPLETION_CHANGED")
242:                 raw = guard_path(self.root / "tasks" / task / pointer["file"]).read_bytes()
243:                 receipt = load(raw)
244:                 plan_path = guard_path(self.root / "tasks" / task / "resolved-plan.json")
245:                 plan_hash = (
246:                     hashlib.sha256(plan_path.read_bytes()).hexdigest()
247:                     if plan_path.exists()
248:                     else None
249:                 )
250:                 if (
251:                     hashlib.sha256(raw).hexdigest() != pointer["sha256"]
252:                     or canonical(receipt) != raw
253:                     or receipt["claim_epoch"] != row["epoch"]
254:                     or receipt["config_hash"] != row["hash"]
255:                     or receipt["core_proofs"] != self._proofs(task)
256:                     or receipt["resolved_plan_sha256"] != plan_hash
257:                 ):
258:                     raise ContractError("TASK_COMPLETION_CHANGED")
259:                 public["result"] = receipt["result"]
260:             except (ContractError, OSError, KeyError, ValueError, TypeError, IndexError) as exc:
261:                 public.update(
262:                     status="INTEGRITY_ERROR",
263:                     error=getattr(exc, "code", type(exc).__name__),
264:                     usable=False,
265:                 )
266:         return public

315:     def _proofs(self, task):
316:         row = self._snapshot(task)
317:         x = load(row["raw"])
318:         slots = self.state.slots(task)
319:         expected = {"R"} if x["kind"] == "research" else {"R", "B"}
320:         if x["kind"] == "replay":
321:             expected = {"E"}
322:         plan_path = guard_path(self.root / "tasks" / task / "resolved-plan.json")
323:         if x["kind"] != "research":
324:             plan = load(plan_path.read_bytes())
325:             expected |= {f"P:{i}" for i in range(plan["decision_count"])}
326:             if (
327:                 plan["task_config_hash"] != row["hash"]
328:                 or set(plan["slots"]) != expected
329:                 or plan["slots"] != {n: s["binding"]["child_run_id"] for n, s in slots.items()}
330:                 or hashlib.sha256(plan["request_utf8"].encode()).hexdigest()
331:                 != plan["request_sha256"]
332:             ):
333:                 raise ContractError("TASK_RESOLVED_PLAN_CHANGED")
334:         if set(slots) != expected:
335:             raise ContractError("TASK_REQUIRED_SLOT_SET_MISMATCH")
336:         proofs = {}
337:         for name, slot in slots.items():
338:             b = slot["binding"]
339:             if (
340:                 b["task_id"] != task
341:                 or b["workspace_id"] != self.state.workspace_id
342:                 or b["plan_hash"] != row["hash"]
343:             ):
344:                 raise ContractError("TASK_SLOT_PARENT_CHANGED")
345:             m, contents = read_core(b)
346:             if m["status"] != "SUCCEEDED":
347:                 raise ContractError("REQUIRED_STAGE_REJECTED")
348:             if name.startswith("P:"):
349:                 parent = slots["B" if x["kind"] == "engineering_pipeline" else "E"]["binding"]
350:                 decision = load(plan["request_utf8"])["decisions"][b["decision_index"]]
351:                 original = load(decision["portfolio_request_utf8"])
352:                 if (
353:                     b["parent_slot_id"] != parent["slot_id"]
354:                     or b["parent_execution_run_id"] != parent["child_run_id"]
355:                     or b["physical_ledger_root"]
356:                     != str(
357:                         guard_path(parent["physical_ledger_root"])
358:                         / parent["child_run_id"]
359:                         / "portfolio-runs"

457:         self._snapshot(task)

459:             raise ContractError("INVALID_TASK_ARTIFACT")

468:     def _verified_financial_prefix(self, binding):
469:         """Current physical financial authority, also for already successful children."""
470:         _, contents = read_core(binding)
471:         result = load(contents["result.json"])
472:         evidence = result.get("committed_state_verification", {})
473:         if evidence.get("financial_state_known") is False:
474:             raise ContractError("TASK_FINANCIAL_RESULT_UNKNOWN")
475:         root = guard_path(binding["physical_ledger_root"]) / binding["child_run_id"]
476:         # Never let TransactionStore create a missing directory and call it an empty prefix.
477:         if (
478:             not guard_path(root / "transactions").is_dir()
479:             or not guard_path(root / "transactions/initial.json").is_file()
480:         ):
481:             raise ContractError("TASK_TRANSACTION_FILES_MISSING")
482:         state, records, files = TransactionStore(root / "transactions").verified_snapshot()
483:         computed = projection(state, records)
484:         if files != load(contents["transaction-files.json"]):
485:             raise ContractError("TASK_FINANCIAL_RESULT_CHANGED")
486:         if binding["service_kind"] == "P5_REPLAY":
487:             comparison = load(contents["comparison.json"])
488:             if (
489:                 result["status"] != "NUMERICAL_REPLAY_VERIFIED"
490:                 or comparison != {"expected": computed, "actual": computed, "equal": True}
491:                 or result["state_sha256"] != computed["state_sha256"]
492:             ):
493:                 raise ContractError("TASK_FINANCIAL_RESULT_CHANGED")
494:         elif result["status"] == "SUCCEEDED":
495:             if any(result[k] != v for k, v in computed.items()):
496:                 raise ContractError("TASK_FINANCIAL_RESULT_CHANGED")
497:         elif (
498:             evidence.get("financial_state_known") is not True
499:             or result.get("last_committed_account") != computed["account"]
500:             or evidence.get("state_sha256") != computed["state_sha256"]
501:             or evidence.get("financial_proof_sha256") != digest(computed["reconciliation"])
502:             or evidence.get("commit_count") != len(records)
503:             or evidence.get("prefix_sha256")
504:             != (records[-1]["sha256"] if records else computed["state_sha256"])
505:         ):
506:             raise ContractError("TASK_FINANCIAL_RESULT_CHANGED")
507:         return result

509:     def _verify_cancelled_finances(self, task):
510:         """No safe cancellation certificate may rely on a cached known-state flag."""
511:         for slot in self.state.slots(task).values():
512:             binding = slot["binding"]
513:             if binding["service_kind"] not in ("P5", "P5_REPLAY"):
514:                 continue
515:             root = guard_path(binding["physical_ledger_root"]) / binding["child_run_id"]
516:             if slot["state"] == "ALLOCATED" and not root.exists():
517:                 continue
518:             self._verified_financial_prefix(binding)
519:             if binding["service_kind"] == "P5_REPLAY":
520:                 x = load(self._snapshot(task)["raw"].encode())
521:                 source = self.state.slots(x["payload"]["source_task"])["B"]["binding"]
522:                 self._verified_financial_prefix(source)

524:     def financial_result(self, task):
525:         row = self._snapshot(task)
526:         self.validate_effective(row["raw"].encode())
527:         slots = self.state.slots(task)
528:         if "B" not in slots:
529:             raise ContractError("NO_BACKTEST_STAGE")
530:         _, contents = read_core(slots["B"]["binding"])
531:         original = load(contents["result.json"])
532:         # Preserve the original opaque failure report for diagnosis; this never
533:         # certifies cash, and cancellation always calls the strict prefix gate.
534:         if (
535:             original["status"] != "SUCCEEDED"

232:     def verify(self, root, run, config=None, *, terminal=True):
233:         b = self.binding()
234:         if str(guard_path(root)) != b["physical_ledger_root"] or run != b["child_run_id"]:
235:             raise ContractError("TASK_CORE_OWNERSHIP_MISMATCH")
236:         ledger = ExperimentLedger(root)
237:         m = ledger.verify(run) if terminal else {"files": []}
238:         lineage = json.loads((ledger.path(run) / "lineage.json").read_bytes())
239:         actual = json.loads((ledger.path(run) / "config.json").read_bytes())
240:         if (
241:             lineage.get("task_binding") != b
242:             or lineage.get("task_binding_hash") != digest(b)
243:             or digest(actual) != b["core_config_hash"]
244:             or (config is not None and actual != config)
245:         ):
246:             raise ContractError("TASK_INITIAL_BINDING_MISMATCH")
247:         if len({e["path"] for e in m["files"]}) != len(m["files"]):
248:             raise ContractError("TASK_DUPLICATE_MANIFEST_MEMBER")
249:         if any(not re.fullmatch(r"[a-z0-9_-]+\.(json|html)", e["path"]) for e in m["files"]):
250:             raise ContractError("TASK_ARTIFACT_BOUNDARY")
251:         initial = read_chain(ledger.path(run) / "events.jsonl")[0]["payload"]
252:         if initial["config_hash"] != digest(actual) or initial["lineage_hash"] != digest(lineage):
253:             raise ContractError("TASK_INITIAL_RECORD_CHANGED")
254:         return m

256:     def existing(self):
257:         b = self.binding()
258:         root = guard_path(b["physical_ledger_root"]) / b["child_run_id"]
259:         slot = self.authority.state.slots(self.authority.task)[self.slot]
260:         if slot["state"] in ("ALLOCATED", "BOUND"):
261:             if root.exists():
262:                 raise ContractError("TASK_UNREGISTERED_CORE_DIRECTORY")
263:             return None
264:         if not root.exists():
265:             raise ContractError("RECOVERY_REQUIRED")
266:         try:
267:             m = self.verify(root.parent, root.name)
268:         except ContractError as exc:
269:             if exc.code == "NONTERMINAL_RUN" and b["service_kind"] == "P4":
270:                 raise ContractError("RECOVERY_REQUIRED") from exc
271:             raise
272:         name = "report.json" if b["service_kind"] == "P3" else "result.json"
273:         raw = guard_path(root / name).read_bytes()
274:         matches = [e for e in m["files"] if e["path"] == name]
275:         if len(matches) != 1 or hashlib.sha256(raw).hexdigest() != matches[0]["sha256"]:
276:             raise ContractError("TASK_RESULT_NOT_MANIFESTED")
277:         result = json.loads(raw)
278:         if b["service_kind"] == "P3":
279:             return {"run_id": root.name, "manifest": m, "report": result}
280:         return {**result, "manifest": m}

284:         parent = self.binding()

44:     values = []

22: @contextmanager
23: def mutex(path, timeout=5):
24:     f = Path(path).open("a+b")
25:     if f.tell() == 0:
26:         f.write(b"0")
27:         f.flush()
28:     start = time.monotonic()
29:     while True:
30:         try:
31:             f.seek(0)
32:             if IS_WINDOWS:
33:                 import msvcrt
34: 
35:                 msvcrt.locking(f.fileno(), msvcrt.LK_NBLCK, 1)
36:             else:
37:                 import fcntl
38: 
39:                 fcntl.flock(f.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
40:             break
41:         except OSError as exc:
42:             if time.monotonic() - start >= timeout:
43:                 f.close()
44:                 raise ContractError("LEDGER_LOCK_TIMEOUT") from exc
45:             time.sleep(0.01)
46:     try:
47:         yield
48:     finally:
49:         f.seek(0)
50:         if IS_WINDOWS:
51:             import msvcrt
52: 
53:             msvcrt.locking(f.fileno(), msvcrt.LK_UNLCK, 1)
54:         else:
55:             import fcntl
56: 
57:             fcntl.flock(f.fileno(), fcntl.LOCK_UN)
58:         f.close()

61: def read_chain(path):
62:     if not Path(path).exists():
63:         return []
64:     previous = "0" * 64
65:     records = []
66:     for index, line in enumerate(Path(path).read_bytes().splitlines()):
67:         item = json.loads(line)
68:         value = {k: v for k, v in item.items() if k != "hash"}
69:         if value["seq"] != index or value["previous"] != previous or digest(value) != item["hash"]:
70:             raise ContractError("LEDGER_CHAIN_CORRUPTED")
71:         previous = item["hash"]
72:         records.append(item)
73:     return records
74: 
75: 
76: def append_chain(path, kind, payload):
77:     records = read_chain(path)
78:     item = {
79:         "seq": len(records),
80:         "recorded_at": datetime.now(UTC).isoformat(timespec="microseconds"),
81:         "previous": records[-1]["hash"] if records else "0" * 64,
82:         "kind": kind,
83:         "payload": payload,
84:     }
85:     item["hash"] = digest(item)
86:     with Path(path).open("ab") as f:
87:         f.write(canonical(item) + b"\n")
88:         f.flush()
89:         os.fsync(f.fileno())
90:     return item

92: 
93: def immutable_write(path, content):
94:     with Path(path).open("xb") as f:
95:         f.write(content)
96:         f.flush()
97:         os.fsync(f.fileno())

203:         events = read_chain(root / "events.jsonl")

211:             or m["status"] != events[-1]["kind"]

===== docs/audits/phase-3/STATISTICAL_CONVENTIONS.md =====
1: # 研究数值与统计口径提案
2: 
3: 所有选择在开发资料中预注册，改变方向/参数/处理链/期限/lag/分组/tie规则均是新实验。以下是待G1批准的工程定义，不是正式因子收益通过阈值。
4: 
5: - 顺序raw→clean(缺失规则及去极值)→standardized→neutralized→final；可选环节明确SKIPPED，不计算标签驱动的处理参数。MAD为median(abs(x-median(x)))，k须配置，MAD=0时invalid而非悄悄放过异常；分位去极值固定linear插值。截面z用sample std(ddof=1)，n<2/零方差invalid；横截面rank用平均并列秩/n，Ts_Rank用窗口平均并列秩/L。
6: - rolling完整session网格、min_periods=L；缺失不压缩时间；Std ddof=1、Correlation成对且非零方差，窗口一证券独立。Delay/Delta只找本证券前k个预定session，不是随意前k条非缺数据。Conditional仅白名单比较/布尔与字段，不eval任意Python。
7: - 中性化截面OLS：截距+industry dummy（固定字典排序第一类为基准）+log正市值；PIT暴露缺失排除并计数，n<=列数、秩亏/无法求解返回invalid及诊断。NumPy lstsq提议固定rcond=1e-12并记录rank/singular values/残差，不能把空residuals数组当零残差。工程参考用独立有理数矩阵，不从待测实现反算。
8: - IC每日对同stage/时点/股票池的有效score/forward label成对过滤，n>=3、双方方差非零。Spearman是平均并列秩后的Pearson。MeanIC有效日等权，std ddof1，ICIR=mean/std默认非年化；至少2日/零std返回null。正值比例只计>0，并另报零值与有效日分母。
9: - 分组按平均并列秩确定bin=min(q-1,floor((rank_avg-1)*q/n))，完整tie块不拆分、security_id仅稳定展示。同分位组过小/空组不强行凑数；min_group_n预注册，TMB需两端合格。等权组标签均值、TMB=top-bottom；单调Spearman及相邻违反数，n不足不报假精度。多期限/年度按固定网格及样本计数，跨因子相关仅同样本交集及stage。
10: - HAC提案：去均值IC序列u，gamma_k=sum(u_t*u_(t-k))/n，S=gamma0+2*sum((1-k/(q+1))*gamma_k,k1..q)，Var(mean)=S/n；显式small_sample_correction=none。若用statsmodels参考必须use_correction=False且固定q，不能混用默认修正/自动lag。q>=h-1且额外自相关lag在开发期冻结。仅在等间隔/样本与假设合格时给渐近正态双侧p=erfc(abs(mean)/sqrt(2*Var(mean)))，明确不是Student有限样本保证；短样本/空洞/零或负有效方差给null+reason。min_hac_n必须在预注册标准中给出并经预审，不凭空默认“足够”。
11: - 确认性family用Holm；探索性可用BY。BH须预先记录适用依赖依据，缺依据不执行。m是观察结果前已登记的全部假设，未完成/失败候选保留（对需保守纳入者记p=1并披露），不能只数赢家或跨family改名避校正；超预算/重复Test/自适应选择触发污染标记，校正不使污染自动消失。
12: - 财务标签按一致拆股/现金红利权益口径；不当作P5账本收益。可见公司行动不得在signal cutoff前注入，标签只在evaluation_cutoff有全部必要信息后生成。现金再投资/税/退市未知规则不虚构，先限明确合成profile，正式规则缺失返回Unsupported/invalid。

26: SQLite提交前宕机，恢复者重新验证整个屏障及completion.json，才可追加恢复Task终态；绝不能从一个恢复的P3/P4终态提升整个Task。SQL终态若指向丢失/损坏回执，保留原记录并对外标INTEGRITY_ERROR/usable=false，不作为成功计数，也不重写旧证据来维持成功。
27: 
28: 这是明确的应用级严格全链成功口径：合法P4风险拒绝仍保留真实P4 REJECTED/FAILED与P5可能SUCCEEDED的原始事实，但该Task以FAILED/REQUIRED_STAGE_REJECTED展示，不能说全链所有必需阶段成功。此规则不改P5风险/事件/金融算法、不把拒绝改成下单、不强制中断原模拟，也不要求成交数或收益为正。所有真实子结果可按各自状态查看。
29: 
30: ## 恢复与未完成阶段
31: 
32: | 观察到的可信状态 | 只允许的恢复动作 | Task不能宣称的事实 |
33: |---|---|---|
34: | R终态已提交、slot映射未写；P4/P5尚未启动 | 核原预分配R绑定后恢复R slot；用其原字节组装/验证PLAN；只有未启动且ALLOCATED的后续slot可执行 | 不能Task成功，不能重跑R、重计或退还试验 |
35: | R/部分P[i]成功，B未形成终态 | 只恢复这些slots；严格满足原P5 resume条件时使用同一B run/请求/上下文和原始事务prefix恢复；否则RECOVERY_REQUIRED | 不能凭已完成P4或缓存余额成功，不能新建B替代 |
36: | R/P[i]成功，B缺失但START_INTENT已登记，或存在半初始化/部分发布 | FAILED/RECOVERY_REQUIRED，保留所有记录；不自动从缺目录推断未执行 | 不能自动重做或抹去已提交副作用 |
37: | B声称SUCCEEDED但缺必需P4记录/归属/原始工件 | FAILED/RECOVERY_REQUIRED；不能在B结束后补开P4来填清单 | 不能接受B的success字符串替代全链 |
38: | B真实FAILED、P[i]失败/拒绝、unknown或真实取消 | 分别记录全部子事实并按失败/安全取消规则归并；未运行后续slot标明未执行 | 不能删掉失败slot来满足all条件 |
39: | 全部必需slots、最终工件及父链成功，但Task回执/索引未提交 | 全量重验证，补完成回执/终态事务；不重新执行核心 | 不能在补验证之前发布成功 |
40: | 全部屏障、回执及终态事务通过 | 返回原Task和原子run引用，后续同key幂等读取 | 不能生成新核心run或覆盖历史 |
41: 
42: 只有slot仍ALLOCATED、无START_INTENT、无核心目录/初始记录且完整枚举和claim证据均能证明从未执行时，才可首次调用。任何枚举错误、未知/冲突、活跃旧claim、配置漂移都失败关闭。P5恢复只适用既有resume_engineering条件（同包/原config/原prefix、无部分发布等）；不能借Task恢复放宽原拒绝。replay是新执行且新归属，resume是同一执行同一slot，不能互换。
43: 
44: G2必须在真实服务/SQLite/磁盘执行R、每个P[i]、B、RV/BV/CV提交及映射前后、completion发布与SQL提交之间的断点，验证未完成任务不成功、无重复核心调用/试验或金融提交，取消竞态和原始失败仍完整。设计模型通过不代表这些实际故障已测。
